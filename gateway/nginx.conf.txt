events {}

http {
    # 10 запросов/сек на один IP; burst разрешает короткий всплеск из 20 запросов.
    limit_req_zone $binary_remote_addr zone=api_limit:10m rate=10r/s;

    upstream order_cluster {
        # Два экземпляра сервиса. Nginx по умолчанию применяет round-robin.
        server order-service-a:8003;
        server order-service-b:8003;
    }

    server {
        listen 80;
        limit_req zone=api_limit burst=20 nodelay;
        limit_req_status 429;

        location = /health {
            default_type application/json;
            return 200 '{"service":"gateway","status":"ok"}';
        }

        # URI переписывается: /api/catalog/items -> /items внутри catalog-service.
        location /api/identity/ { proxy_pass http://identity-service:8001/; }
        location /api/catalog/ { proxy_pass http://catalog-service:8002/; }
        location /api/orders/ { proxy_pass http://order_cluster/; }
        location /api/kitchen/ { proxy_pass http://kitchen-service:8004/; }
        location /api/notifications/ { proxy_pass http://notification-service:8005/; }
        location /api/analytics/ { proxy_pass http://analytics-service:8006/; }

        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Request-ID $request_id;
    }
}
